var basins = ee.FeatureCollection('WWF/HydroSHEDS/v1/Basins/hybas_5');
var DRIVE_FOLDER = 'iran war project';

var rows = SITES.map(function (s) {
  var pt = ee.Geometry.Point([s[1], s[2]]);
  var b  = ee.Feature(basins.filterBounds(pt).first());
  var id = b.get('HYBAS_ID');

  var interior5  = b.geometry().buffer(-5000, 1000);
  var interior10 = b.geometry().buffer(-10000, 1000);

  return ee.Feature(null, {
    site: s[0], waterbody_type: s[3], lon: s[1], lat: s[2],
    hybas_id: id,
    sub_area_km2: b.get('SUB_AREA'),
    up_area_km2: b.get('UP_AREA'),
    next_down: b.get('NEXT_DOWN'),
    main_bas: b.get('MAIN_BAS'),
    endo: b.get('ENDO'),
    interior_5km:  ee.Algorithms.If(interior5.contains(pt, 1), 1, 0),
    interior_10km: ee.Algorithms.If(interior10.contains(pt, 1), 1, 0)
  });
});

var fc = ee.FeatureCollection(rows);
print('Reservoir to basin assignment, 27 sites:');
print(fc);

Export.table.toDrive({
  collection: fc,
  description: 'Iran_site_basin_assignment',
  folder: DRIVE_FOLDER,
  fileNamePrefix: 'Iran_site_basin_assignment',
  fileFormat: 'CSV',
  selectors: ['site','waterbody_type','lon','lat','hybas_id','sub_area_km2',
              'up_area_km2','next_down','main_bas','endo',
              'interior_5km','interior_10km']
});

var up = ee.FeatureCollection(SITES.map(function (s) {
  var pt = ee.Geometry.Point([s[1], s[2]]);
  var id = ee.Feature(basins.filterBounds(pt).first()).get('HYBAS_ID');
  var lvl1 = basins.filter(ee.Filter.eq('NEXT_DOWN', id));
  var lvl2 = basins.filter(ee.Filter.inList('NEXT_DOWN',
               lvl1.aggregate_array('HYBAS_ID')));
  return lvl1.merge(lvl2).map(function (f) {
    return ee.Feature(null, {
      site: s[0], hybas_id_site: id,
      hybas_id_upstream: f.get('HYBAS_ID'),
      upstream_area_km2: f.get('SUB_AREA')
    });
  });
})).flatten();

Export.table.toDrive({
  collection: up,
  description: 'Iran_site_basin_upstream',
  folder: DRIVE_FOLDER,
  fileNamePrefix: 'Iran_site_basin_upstream',
  fileFormat: 'CSV',
  selectors: ['site','hybas_id_site','hybas_id_upstream','upstream_area_km2']
});
