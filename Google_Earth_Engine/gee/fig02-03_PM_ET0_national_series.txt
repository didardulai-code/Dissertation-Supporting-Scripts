var RUN_MODE = 'export';

var TEST_SEASON = 'JJA';
var TEST_YEAR   = 2025;

var ONLY_SEASONS = null;

var CHUNK_YEARS = 1;

var FIRST_YEAR = 1991;
var DRIVE_FOLDER = 'Iran_PM_ET0';

var ERA5_SCALE = 11132;
var CHIRPS_SCALE = 5566;

var TILE_SCALE = 4;

var SEASONS = [
  {name: 'SON',  startMonth: 9,  endMonth: 11, spans: false, lastYear: 2025},
  {name: 'DJF',  startMonth: 12, endMonth: 2,  spans: true,  lastYear: 2026},
  {name: 'MAM',  startMonth: 3,  endMonth: 5,  spans: false, lastYear: 2026},
  {name: 'JJA',  startMonth: 6,  endMonth: 8,  spans: false, lastYear: 2025},
  {name: 'June', startMonth: 6,  endMonth: 6,  spans: false, lastYear: 2026}
];

var iran = ee.FeatureCollection('FAO/GAUL/2015/level0')
  .filter(ee.Filter.eq('ADM0_NAME', 'Iran  (Islamic Republic of)'));

var geom = iran.geometry();

var era5 = ee.ImageCollection('ECMWF/ERA5_LAND/DAILY_AGGR')
  .select([
    'temperature_2m',
    'temperature_2m_min',
    'temperature_2m_max',
    'surface_pressure',
    'surface_solar_radiation_downwards_sum',
    'potential_evaporation_sum'
  ]);

var era5Hourly = ee.ImageCollection('ECMWF/ERA5_LAND/HOURLY')
  .select([
    'dewpoint_temperature_2m',
    'u_component_of_wind_10m',
    'v_component_of_wind_10m'
  ]);

var chirps = ee.ImageCollection('UCSB-CHG/CHIRPS/DAILY')
  .select('precipitation');

var ERA5_PROJ = era5.first().select('temperature_2m').projection();

var PHI = ee.Image.pixelLonLat()
  .select('latitude')
  .multiply(Math.PI / 180.0)
  .setDefaultProjection(ERA5_PROJ);

var ELEV = ee.Image('USGS/SRTMGL1_003')
  .select('elevation')
  .resample('bilinear')
  .reproject({crs: ERA5_PROJ})
  .rename('elevation_m');

era5.aggregate_max('system:time_start').evaluate(function (t, err) {
  if (err) {
    print('ERA5-Land DAILY_AGGR last-date lookup failed: ' + err);
    return;
  }
  print('ERA5-Land DAILY_AGGR last image: ' +
        new Date(t).toISOString().slice(0, 10));
});

era5Hourly.aggregate_max('system:time_start').evaluate(function (t, err) {
  if (err) {
    print('ERA5-Land HOURLY last-date lookup failed: ' + err);
    return;
  }
  print('ERA5-Land HOURLY last image: ' +
        new Date(t).toISOString().slice(0, 13) + ':00 UTC');
});

chirps.aggregate_max('system:time_start').evaluate(function (t, err) {
  if (err) {
    print('CHIRPS DAILY last-date lookup failed: ' + err);
    return;
  }
  print('CHIRPS DAILY last image: ' +
        new Date(t).toISOString().slice(0, 10));
});

function extraterrestrialRadiationMJ(doy) {
  var J = ee.Number(doy);
  var b = J.multiply(2 * Math.PI / 365.0);

  var dr = ee.Image.constant(
    b.cos().multiply(0.033).add(1)
  );

  var dec = ee.Image.constant(
    b.subtract(1.39).sin().multiply(0.409)
  );

  var x = PHI.tan()
    .multiply(dec.tan())
    .multiply(-1)
    .clamp(-1, 1);

  var ws = x.acos();

  return ws.multiply(PHI.sin()).multiply(dec.sin())
    .add(
      PHI.cos()
        .multiply(dec.cos())
        .multiply(ws.sin())
    )
    .multiply(dr)
    .multiply(0.0820)
    .multiply(24 * 60 / Math.PI)
    .rename('Ra_MJ');
}

function saturationVapourPressure(t) {
  return t.multiply(17.27)
    .divide(t.add(237.3))
    .exp()
    .multiply(0.6108);
}

function dailyPenmanMonteith(img, hourlySeason) {
  var date = ee.Date(img.get('system:time_start'));
  var nextDate = date.advance(1, 'day');
  var h = hourlySeason.filterDate(date, nextDate);

  var Tmax = img.select('temperature_2m_max').subtract(273.15);
  var Tmin = img.select('temperature_2m_min').subtract(273.15);

  var T = Tmax.add(Tmin).multiply(0.5);

  var es = saturationVapourPressure(Tmax)
    .add(saturationVapourPressure(Tmin))
    .multiply(0.5);

  var ea = h.map(function (hourlyImg) {
    var Tdew = hourlyImg.select('dewpoint_temperature_2m')
      .subtract(273.15);

    return saturationVapourPressure(Tdew).rename('ea_kpa');
  }).mean();

  var vpd = es.subtract(ea)
    .max(0)
    .rename('vpd_k');

  var delta = saturationVapourPressure(T)
    .multiply(4098)
    .divide(T.add(237.3).pow(2));

  var gamma = img.select('surface_pressure')
    .divide(1000)
    .multiply(0.000665);

  var wind10to2 = 4.87 / Math.log((67.8 * 10) - 5.42);

  var u2 = h.map(function (hourlyImg) {
    var u10 = hourlyImg.select('u_component_of_wind_10m');
    var v10 = hourlyImg.select('v_component_of_wind_10m');

    var wind10 = u10.pow(2)
      .add(v10.pow(2))
      .sqrt();

    return wind10
      .multiply(wind10to2)
      .rename('u2_ms');
  }).mean();

  var Rs = img.select('surface_solar_radiation_downwards_sum')
    .divide(1e6)
    .max(0);

  var doy = date.getRelative('day', 'year').add(1);
  var Ra = extraterrestrialRadiationMJ(doy);

  var Rso = ELEV.multiply(2e-5)
    .add(0.75)
    .multiply(Ra)
    .max(1e-6);

  var Rns = Rs.multiply(0.77);

  var TmaxK = Tmax.add(273.16);
  var TminK = Tmin.add(273.16);

  var temperatureTerm = TmaxK.pow(4)
    .add(TminK.pow(4))
    .multiply(0.5);

  var humidityTerm = ea.sqrt()
    .multiply(-0.14)
    .add(0.34);

  var relativeSolarRadiation = Rs.divide(Rso)
    .min(1.0);

  var cloudinessTerm = relativeSolarRadiation
    .multiply(1.35)
    .subtract(0.35);

  var Rnl = temperatureTerm
    .multiply(4.903e-9)
    .multiply(humidityTerm)
    .multiply(cloudinessTerm);

  var Rn = Rns.subtract(Rnl)
    .rename('rn_mj');

  var G = ee.Image.constant(0);

  var tterm = ee.Image.constant(900)
    .divide(T.add(273));

  var num = delta.multiply(0.408)
    .multiply(Rn.subtract(G))
    .add(
      gamma.multiply(tterm)
        .multiply(u2)
        .multiply(vpd)
    );

  var den = delta.add(
    gamma.multiply(
      u2.multiply(0.34).add(1)
    )
  );

  var et0 = num.divide(den)
    .max(0)
    .rename('et0');

  return et0
    .addBands(Rn.rename('rn_mj'))
    .addBands(u2.rename('u2_ms'))
    .addBands(vpd.rename('vpd_k'))
    .copyProperties(img, ['system:time_start']);
}

function seasonWindow(s, y) {
  var start = s.spans
    ? ee.Date.fromYMD(y - 1, s.startMonth, 1)
    : ee.Date.fromYMD(y, s.startMonth, 1);

  var end = ee.Date.fromYMD(y, s.endMonth, 1)
    .advance(1, 'month');

  return {start: start, end: end};
}

function rowFor(s, y) {
  var w = seasonWindow(s, y);

  var e5 = era5.filterDate(w.start, w.end);
  var e5h = era5Hourly.filterDate(w.start, w.end);
  var ch = chirps.filterDate(w.start, w.end);

  var pmDaily = e5.map(function (img) {
    return dailyPenmanMonteith(img, e5h);
  });

  var pet_pm = pmDaily.select('et0')
    .sum()
    .rename('pet_pm_mm');

  var et0_daily_mean = pmDaily.select('et0')
    .mean()
    .rename('et0_daily_mean');

  var rn_mj = pmDaily.select('rn_mj')
    .mean()
    .rename('rn_mj');

  var u2_ms = pmDaily.select('u2_ms')
    .mean()
    .rename('u2_ms');

  var vpd_k = pmDaily.select('vpd_k')
    .mean()
    .rename('vpd_k');

  var tmean = e5.select('temperature_2m')
    .mean()
    .subtract(273.15)
    .rename('tmean_C');

  var tmax = e5.select('temperature_2m_max')
    .mean()
    .subtract(273.15)
    .rename('tmax_C');

  var tmin = e5.select('temperature_2m_min')
    .mean()
    .subtract(273.15)
    .rename('tmin_C');

  var precip = ch.sum()
    .rename('precip_chirps_mm');

  var era5Stats = tmean
    .addBands(tmax)
    .addBands(tmin)
    .addBands(pet_pm)
    .addBands(et0_daily_mean)
    .addBands(rn_mj)
    .addBands(u2_ms)
    .addBands(vpd_k)
    .reduceRegion({
      reducer: ee.Reducer.mean(),
      geometry: geom,
      scale: ERA5_SCALE,
      maxPixels: 1e13,
      tileScale: TILE_SCALE,
      bestEffort: false
    });

  var chirpsStats = precip.reduceRegion({
    reducer: ee.Reducer.mean(),
    geometry: geom,
    scale: CHIRPS_SCALE,
    maxPixels: 1e13,
    tileScale: TILE_SCALE,
    bestEffort: false
  });

  var expected = w.end.difference(w.start, 'day');
  var hoursExpected = expected.multiply(24);

  var nE5 = e5.size();
  var nE5h = e5h.size();
  var nCh = ch.size();

  var complete = ee.Number(nE5).eq(expected)
    .and(ee.Number(nE5h).eq(hoursExpected))
    .and(ee.Number(nCh).eq(expected));

  return ee.Feature(null, {
    season:              s.name,
    season_year:         y,
    start_date:          w.start.format('YYYY-MM-dd'),
    end_date:            w.end.format('YYYY-MM-dd'),
    days_expected:       expected,
    hours_expected:      hoursExpected,
    n_era5_days:         nE5,
    n_era5_hours:        nE5h,
    n_chirps_days:       nCh,
    qc_complete:         complete,
    tmean_C:             era5Stats.get('tmean_C'),
    tmax_C:              era5Stats.get('tmax_C'),
    tmin_C:              era5Stats.get('tmin_C'),
    pet_pm_mm:           era5Stats.get('pet_pm_mm'),
    et0_daily_mean:      era5Stats.get('et0_daily_mean'),
    rn_mj:               era5Stats.get('rn_mj'),
    u2_ms:               era5Stats.get('u2_ms'),
    vpd_k:               era5Stats.get('vpd_k'),
    precip_chirps_mm:    chirpsStats.get('precip_chirps_mm')
  });
}

var COLUMNS = [
  'season',
  'season_year',
  'start_date',
  'end_date',
  'days_expected',
  'hours_expected',
  'n_era5_days',
  'n_era5_hours',
  'n_chirps_days',
  'qc_complete',
  'tmean_C',
  'tmax_C',
  'tmin_C',
  'pet_pm_mm',
  'et0_daily_mean',
  'rn_mj',
  'u2_ms',
  'vpd_k',
  'precip_chirps_mm'
];

if (RUN_MODE === 'test') {
  var testSeason = null;

  SEASONS.forEach(function (s) {
    if (s.name === TEST_SEASON) {
      testSeason = s;
    }
  });

  if (testSeason === null) {
    print('TEST_SEASON "' + TEST_SEASON + '" is not in the SEASONS list.');
  } else {
    print('TEST MODE - computing ' + TEST_SEASON + ' ' + TEST_YEAR +
          ' only. No export tasks will be created.');

    var fmt = function (v, d) {
      return (v === null || v === undefined)
        ? 'null'
        : Number(v).toFixed(d);
    };

    rowFor(testSeason, TEST_YEAR).evaluate(function (f, err) {
      if (err) {
        print('Test row failed: ' + err);
        return;
      }

      var p = f.properties;

      print('  window              ' + p.start_date + ' to ' + p.end_date);
      print('  days expected       ' + p.days_expected);
      print('  ERA5 daily found    ' + p.n_era5_days);
      print('  hours expected      ' + p.hours_expected);
      print('  ERA5 hourly found   ' + p.n_era5_hours);
      print('  CHIRPS days found   ' + p.n_chirps_days);
      print('  qc_complete         ' + p.qc_complete +
            (p.qc_complete === 1
              ? '  (all records cover the window)'
              : '  <- INCOMPLETE, check source dates and lastYear'));

      print('  Tmean / Tmax / Tmin ' +
            fmt(p.tmean_C, 2) + ' / ' +
            fmt(p.tmax_C, 2) + ' / ' +
            fmt(p.tmin_C, 2) + ' degC');

      print('  ET0 PM seasonal     ' + fmt(p.pet_pm_mm, 1) + ' mm');
      print('  ET0 PM daily mean   ' + fmt(p.et0_daily_mean, 2) + ' mm/day');
      print('  mean Rn             ' + fmt(p.rn_mj, 2) + ' MJ m-2 day-1');
      print('  mean u2             ' + fmt(p.u2_ms, 2) + ' m/s');
      print('  mean VPD            ' + fmt(p.vpd_k, 2) + ' kPa');

      print('  CHIRPS precip       ' + fmt(p.precip_chirps_mm, 1) + ' mm');

      print('  P - ET0 PM          ' +
            fmt(p.precip_chirps_mm - p.pet_pm_mm, 1) +
            ' mm  (computed here for inspection only, not exported)');

      print('Test complete. Set RUN_MODE = "export" to create the tasks.');
    });
  }
}

if (RUN_MODE === 'export') {
  var nTasks = 0;
  var nRows = 0;

  SEASONS.forEach(function (s) {
    if (ONLY_SEASONS !== null &&
        ONLY_SEASONS.indexOf(s.name) === -1) {
      return;
    }

    var years = [];

    for (var y = FIRST_YEAR; y <= s.lastYear; y++) {
      years.push(y);
    }

    var blockSize = (CHUNK_YEARS > 0)
      ? CHUNK_YEARS
      : years.length;

    for (var i = 0; i < years.length; i += blockSize) {
      var block = years.slice(i, i + blockSize);

      var rows = [];
      for (var k = 0; k < block.length; k++) {
        rows.push(rowFor(s, block[k]));
      }

      var fc = ee.FeatureCollection(rows);

      var suffix = (CHUNK_YEARS > 0)
        ? '_' + block[0] + '_' + block[block.length - 1]
        : '_' + FIRST_YEAR + '_' + s.lastYear;

      var name = 'Iran_seasonal_T_PET_CHIRPS_' + s.name + suffix;

      Export.table.toDrive({
        collection: fc,
        description: name,
        folder: DRIVE_FOLDER,
        fileNamePrefix: name,
        fileFormat: 'CSV',
        selectors: COLUMNS
      });

      nTasks += 1;
      nRows += block.length;

      print('Task created: ' + name +
            '   (' + block.length + ' season-years)');
    }
  });

  print('---');
  print(nTasks + ' export tasks covering ' + nRows + ' season-years.');
  print('Run the tasks from the Tasks tab. If a task fails, reduce CHUNK_YEARS ' +
        'or rerun only the affected season with ONLY_SEASONS.');
  print('Concatenate the CSVs in Python, then filter on qc_complete == 1.');
}
