var MAP_TARGETS = [
  {season: 'SON',    year: 2025, label: 'SON_2025'},
  {season: 'DJF',    year: 2026, label: 'DJF_2025_26'},
  {season: 'MAM',    year: 2026, label: 'MAM_2026'},
  {season: 'June',   year: 2026, label: 'June_2026'},
  {season: 'SepJun', year: 2025, label: 'SepJun_2025_26'}
];

var EXPORT_SERIES  = true;
var EXPORT_MONTHLY = true;
var EXPORT_MAPS    = true;

var MONTHLY_FIRST_YEAR   = 2015;
var MONTHLY_CHUNK_YEARS  = 2;

var MONTHLY_FROZEN = false;

var RUN_CONSOLE_CHECKS = false;

var SMAP_START_DATE = '2015-03-31';
var FIRST_COMPLETE = {
  SON: 2015, DJF: 2016, MAM: 2016, JJA: 2015, June: 2016, SepJun: 2015
};
var LAST_YEAR = 2026;

var CLIM_LAST = 2025;

var SMAP_SCALE  = 9000;
var GLDAS_SCALE = 27830;
var TILE_SCALE  = 16;

var DRIVE_FOLDER = 'iran war project';

var FROZEN_HATCH_THRESHOLD = 0.20;

var iran = ee.FeatureCollection('FAO/GAUL/2015/level0')
             .filter(ee.Filter.eq('ADM0_NAME', 'Iran  (Islamic Republic of)'));
var geom = iran.geometry();

var smap  = ee.ImageCollection('NASA/SMAP/SPL4SMGP/008')
              .select(['sm_surface', 'sm_rootzone', 'soil_temp_layer1']);
var gldas = ee.ImageCollection('NASA/GLDAS/V021/NOAH/G025/T3H')
              .select(['RootMoist_inst', 'SoilMoi0_10cm_inst']);

function zerosLike(list) {
  var z = [];
  for (var i = 0; i < list.length; i++) { z.push(0); }
  return z;
}

function frozenImg(img) {
  return img.select('soil_temp_layer1').lt(273.15);
}

function window_(season, year) {
  if (season === 'SON')    return {s: ee.Date.fromYMD(year, 9, 1),
                                   e: ee.Date.fromYMD(year, 12, 1)};
  if (season === 'DJF')    return {s: ee.Date.fromYMD(year - 1, 12, 1),
                                   e: ee.Date.fromYMD(year, 3, 1)};
  if (season === 'MAM')    return {s: ee.Date.fromYMD(year, 3, 1),
                                   e: ee.Date.fromYMD(year, 6, 1)};
  if (season === 'JJA')    return {s: ee.Date.fromYMD(year, 6, 1),
                                   e: ee.Date.fromYMD(year, 9, 1)};
  if (season === 'June')   return {s: ee.Date.fromYMD(year, 6, 1),
                                   e: ee.Date.fromYMD(year, 7, 1)};
  if (season === 'SepJun') return {s: ee.Date.fromYMD(year, 9, 1),
                                   e: ee.Date.fromYMD(year + 1, 7, 1)};
  throw new Error('unknown season ' + season);
}

function areaWeightedMean(img, scale) {
  var area  = ee.Image.pixelArea();
  var names = img.bandNames();
  var areas = img.mask().multiply(area).rename(
    names.map(function (n) { return ee.String(n).cat('_area'); }));
  var sums = img.multiply(area).addBands(areas).reduceRegion({
    reducer: ee.Reducer.sum(), geometry: geom, scale: scale,
    maxPixels: 1e13, tileScale: TILE_SCALE, bestEffort: false});
  return ee.Dictionary.fromLists(names, names.map(function (n) {
    var nn = ee.String(n);
    return ee.Number(sums.get(nn)).divide(ee.Number(sums.get(nn.cat('_area'))));
  }));
}

function seasonMean(coll, season, year) {
  var w = window_(season, year);
  return coll.filterDate(w.s, w.e).mean();
}

function frozenFraction(season, year) {
  var w = window_(season, year);
  return smap.filterDate(w.s, w.e).map(frozenImg).mean().rename('frozen_frac');
}

function seasonCount(coll, season, year) {
  var w = window_(season, year);
  return coll.filterDate(w.s, w.e).size();
}

function buildSeries(coll, seasons, firstYearFn, scale, valueBands) {
  var feats = [];
  seasons.forEach(function (season) {
    for (var y = firstYearFn(season); y <= LAST_YEAR; y++) {
      var w = window_(season, y);

      var n = seasonCount(coll, season, y);

      var img = seasonMean(coll, season, y).select(valueBands);
      var safe = ee.Image(ee.Algorithms.If(n.gt(0), img,
        ee.Image.constant(zerosLike(valueBands))
          .rename(valueBands).updateMask(ee.Image.constant(0))));
      var stats = areaWeightedMean(safe, scale);
      var f = ee.Feature(null, {
        season: season,
        season_year: y,
        n_images: n,
        days_expected: w.e.difference(w.s, 'day'),
        start_date: w.s.format('YYYY-MM-dd'),
        end_date: w.e.format('YYYY-MM-dd')
      });
      for (var bi = 0; bi < valueBands.length; bi++) {
        f = f.set(valueBands[bi], stats.get(valueBands[bi]));
      }
      feats.push(f);
    }
  });
  return ee.FeatureCollection(feats);
}

if (EXPORT_SERIES) {

  var SEASONS = ['SON', 'DJF', 'MAM', 'JJA', 'June', 'SepJun'];

  SEASONS.forEach(function (season) {
    var smapTable = buildSeries(smap, [season],
                                function (s) { return FIRST_COMPLETE[s]; },
                                SMAP_SCALE, ['sm_surface', 'sm_rootzone']);
    var sName = 'Iran_SMAP_seasonal_national_' + season;
    Export.table.toDrive({
      collection: smapTable, description: sName,
      folder: DRIVE_FOLDER, fileNamePrefix: sName,
      fileFormat: 'CSV',
      selectors: ['season', 'season_year', 'sm_surface', 'sm_rootzone',
                  'n_images', 'days_expected', 'start_date', 'end_date']
    });
    print('Task created: ' + sName);

    var gldasTable = buildSeries(gldas, [season],
                                 function () { return 2000; },
                                 GLDAS_SCALE, ['RootMoist_inst', 'SoilMoi0_10cm_inst']);
    var gName = 'Iran_GLDAS_seasonal_national_' + season;
    Export.table.toDrive({
      collection: gldasTable, description: gName,
      folder: DRIVE_FOLDER, fileNamePrefix: gName,
      fileFormat: 'CSV',
      selectors: ['season', 'season_year', 'RootMoist_inst',
                  'SoilMoi0_10cm_inst', 'n_images', 'days_expected',
                  'start_date', 'end_date']
    });
    print('Task created: ' + gName);
  });
}

if (EXPORT_MONTHLY) {
  var chunkStart = MONTHLY_FIRST_YEAR;
  while (chunkStart <= LAST_YEAR) {
    var chunkEnd = Math.min(chunkStart + MONTHLY_CHUNK_YEARS - 1, LAST_YEAR);
    var mFeats = [];

    for (var my = chunkStart; my <= chunkEnd; my++) {
      for (var mm = 1; mm <= 12; mm++) {
        var ms = ee.Date.fromYMD(my, mm, 1);
        var me = ms.advance(1, 'month');
        var mcoll = smap.filterDate(ms, me);
        var mn = mcoll.size();

        var bands = ['sm_surface', 'sm_rootzone'];
        var mimg = mcoll.select(bands).mean();
        if (MONTHLY_FROZEN) {
          mimg = mimg.addBands(mcoll.map(frozenImg).mean().rename('frozen_frac'));
          bands = bands.concat(['frozen_frac']);
        }

        var zeros = zerosLike(bands);
        var safe = ee.Image(ee.Algorithms.If(
          mn.gt(0), mimg,
          ee.Image.constant(zeros).rename(bands)
            .updateMask(ee.Image.constant(0))));

        var mstats = areaWeightedMean(safe, SMAP_SCALE);
        var f = ee.Feature(null, {
          year: my, month: mm, n_images: mn,
          start_date: ms.format('YYYY-MM-dd')
        });
        for (var bj = 0; bj < bands.length; bj++) {
          f = f.set(bands[bj], mstats.get(bands[bj]));
        }
        mFeats.push(f);
      }
    }

    var tag = 'Iran_SMAP_monthly_national_' + chunkStart + '_' + chunkEnd;
    Export.table.toDrive({
      collection: ee.FeatureCollection(mFeats),
      description: tag, folder: DRIVE_FOLDER, fileNamePrefix: tag,
      fileFormat: 'CSV',
      selectors: ['year', 'month', 'sm_surface', 'sm_rootzone']
        .concat(MONTHLY_FROZEN ? ['frozen_frac'] : [])
        .concat(['n_images', 'start_date'])
    });
    print('Task created: ' + tag);
    chunkStart = chunkEnd + 1;
  }
}

function buildAnomaly(season, year) {
  var surfs = [], roots = [];
  for (var y = FIRST_COMPLETE[season]; y <= CLIM_LAST; y++) {

    if (y === year) { continue; }
    var m = seasonMean(smap, season, y);
    surfs.push(m.select('sm_surface'));
    roots.push(m.select('sm_rootzone'));
  }
  if (surfs.length < 5) {
    throw new Error('only ' + surfs.length + ' climatology seasons for '
                    + season + ' - too few to average');
  }
  var surfClim = ee.ImageCollection(surfs).mean().rename('surf_clim');
  var rootClim = ee.ImageCollection(roots).mean().rename('root_clim');

  var obs  = seasonMean(smap, season, year);
  var surf = obs.select('sm_surface').rename('surf');
  var root = obs.select('sm_rootzone').rename('root');

  var frz = frozenFraction(season, year);
  var frzs = [];
  for (var fy = FIRST_COMPLETE[season]; fy <= CLIM_LAST; fy++) {
    if (fy === year) { continue; }
    frzs.push(frozenFraction(season, fy));
  }
  var frzClim = ee.ImageCollection(frzs).mean().rename('frozen_frac_clim');

  return surf.addBands(surfClim)
             .addBands(surf.subtract(surfClim).rename('surf_anom'))
             .addBands(root).addBands(rootClim)
             .addBands(root.subtract(rootClim).rename('root_anom'))
             .addBands(frz).addBands(frzClim)
             .toFloat().clip(geom);
}

if (EXPORT_MAPS) {
  MAP_TARGETS.forEach(function (t) {
    var img  = buildAnomaly(t.season, t.year);
    var name = 'Iran_SMAP_' + t.label + '_anomaly';
    Export.image.toDrive({
      image: img, description: name, folder: DRIVE_FOLDER,
      fileNamePrefix: name, region: geom, scale: SMAP_SCALE,
      crs: 'EPSG:4326', maxPixels: 1e13, fileFormat: 'GeoTIFF'
    });
    print('Task created: ' + name);

    if (RUN_CONSOLE_CHECKS) {
      areaWeightedMean(img.select(['surf_anom', 'root_anom', 'frozen_frac']),
                       SMAP_SCALE)
        .evaluate(function (v, err) {
          if (err) { print(name + ' check failed: ' + err); return; }
          print('--- ' + name + ' national ---');
          print('  surface anomaly    ' + v.surf_anom.toFixed(4) + ' m3/m3'
                + '   = ' + (v.surf_anom * 50).toFixed(2) + ' mm equivalent');
          print('  root zone anomaly  ' + v.root_anom.toFixed(4) + ' m3/m3'
                + '   = ' + (v.root_anom * 1000).toFixed(2) + ' mm equivalent');
          print('  mean fraction with layer-1 soil temp < 0 C: '
                + (v.frozen_frac * 100).toFixed(1) + ' %');
        });

      img.select('frozen_frac').gt(FROZEN_HATCH_THRESHOLD)
         .multiply(ee.Image.pixelArea()).rename('a')
         .addBands(ee.Image.pixelArea().rename('tot'))
         .reduceRegion({reducer: ee.Reducer.sum(), geometry: geom,
                        scale: SMAP_SCALE, maxPixels: 1e13,
                        tileScale: TILE_SCALE})
         .evaluate(function (v, err) {
           if (err) { return; }
           print('  layer-1 soil temp < 0 C for more than '
                 + (FROZEN_HATCH_THRESHOLD * 100).toFixed(0) + '% of the season: '
                 + (100 * v.a / v.tot).toFixed(1) + '% of Iran');
         });
    }
  });
}
