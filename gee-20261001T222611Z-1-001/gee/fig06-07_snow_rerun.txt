var EXPORT_BASIN_SERIES = true;
var EXPORT_BAND_SERIES  = true;
var EXPORT_BASIN_SCDLSC = true;
var EXPORT_BAND_SCDLSC  = false;

var Y0 = 2003;
var Y1 = 2026;
var SNOW_START_MONTH = 10;
var SNOW_END_MONTH   = 7;
var WINDOW_DAYS = 8;
var SCALE = 1000;
var TILE_SCALE = 16;
var DRIVE_FOLDER = 'iran_snow_rerun';

var THRESH_PRIMARY = 40;
var THRESH_SENS    = 10;

var ELEV_EDGES = [0, 500, 1000, 1500, 2000, 2500, 3000, 9000];

var MIN_BASIN_AREA_KM2 = 500;
var MIN_MEAN_ELEV_M    = 800;

var iran = ee.FeatureCollection('USDOS/LSIB_SIMPLE/2017')
             .filter(ee.Filter.eq('country_na', 'Iran')).first().geometry();

var dem = ee.Image('USGS/SRTMGL1_003').select('elevation')
            .resample('bilinear').reproject({crs: 'EPSG:4326', scale: SCALE})
            .rename('elev');

var bandIdx = ee.Image(0).rename('eb');
for (var bi = 1; bi < ELEV_EDGES.length - 1; bi++) {
  bandIdx = bandIdx.where(dem.gte(ELEV_EDGES[bi]), bi);
}
bandIdx = bandIdx.toInt().rename('eb');

var landMask = ee.ImageCollection('MODIS/006/MOD44W')
                 .sort('system:time_start', false).first()
                 .select('water_mask').eq(0).rename('land');

var terra = ee.ImageCollection('MODIS/061/MOD10A1').select('NDSI_Snow_Cover');
var aqua  = ee.ImageCollection('MODIS/061/MYD10A1').select('NDSI_Snow_Cover');

function validNDSI(im) {
  var v = im.select('NDSI_Snow_Cover');
  return v.updateMask(v.gte(0).and(v.lte(100)));
}

function windowComposite(d0, d1) {
  var coll = terra.filterDate(d0, d1).merge(aqua.filterDate(d0, d1)).map(validNDSI);
  var med  = coll.median().updateMask(landMask);
  var obs  = med.mask().rename('obs');
  var s40  = med.gte(THRESH_PRIMARY).rename('s40');
  var s10  = med.gte(THRESH_SENS).rename('s10');
  return ee.Image.cat([obs, s40, s10]).unmask(0).toByte();
}

function snowYearWindows(Y) {
  var start = ee.Date.fromYMD(Y - 1, SNOW_START_MONTH, 1);
  var end   = ee.Date.fromYMD(Y,     SNOW_END_MONTH,   1);
  var n     = end.difference(start, 'day').divide(WINDOW_DAYS).ceil();
  return ee.List.sequence(0, n.subtract(1)).map(function (i) {
    var d0 = start.advance(ee.Number(i).multiply(WINDOW_DAYS), 'day');
    var d1 = d0.advance(WINDOW_DAYS, 'day');
    d1 = ee.Date(ee.Algorithms.If(d1.millis().gt(end.millis()), end, d1));
    var img = windowComposite(d0, d1)
                .set('Y', Y)
                .set('w', i)
                .set('doy', ee.Number(i).multiply(WINDOW_DAYS))
                .set('wdays', d1.difference(d0, 'day'))
                .set('system:time_start', d0.millis());
    return img;
  });
}

var basins = ee.FeatureCollection('WWF/HydroSHEDS/v1/Basins/hybas_5')
  .filterBounds(iran)
  .filter(ee.Filter.gte('SUB_AREA', MIN_BASIN_AREA_KM2))
  .map(function (f) {
    var g = f.geometry().intersection(iran, 100);
    var z = dem.reduceRegion({
      reducer: ee.Reducer.mean().combine(ee.Reducer.percentile([10, 50, 90]), '', true),
      geometry: g, scale: 1000, maxPixels: 1e10, tileScale: TILE_SCALE});
    return f.setGeometry(g).set({iran_area_km2: g.area(100).divide(1e6),
                  mean_elev_m: z.get('elev_mean'),
                  elev_p10: z.get('elev_p10'), elev_p50: z.get('elev_p50'),
                  elev_p90: z.get('elev_p90')});
  })
  .filter(ee.Filter.gte('iran_area_km2', MIN_BASIN_AREA_KM2))
  .filter(ee.Filter.gte('mean_elev_m', MIN_MEAN_ELEV_M));
print(ee.String('basins carried forward: ').cat(ee.Number(basins.size()).format()));

var AREA = ee.Image.pixelArea();

function seriesFeatures(Y) {
  var wins = snowYearWindows(Y);
  return ee.FeatureCollection(wins.map(function (imgU) {
    var img = ee.Image(imgU);
    var a   = img.multiply(AREA);
    var props = {Y: Y, window: img.get('w'), doy: img.get('doy'),
                 wdays: img.get('wdays')};

    var out = ee.FeatureCollection([]);

    if (EXPORT_BASIN_SERIES) {
      var perB = a.reduceRegions({collection: basins, reducer: ee.Reducer.sum(),
                                  scale: SCALE, tileScale: TILE_SCALE})
        .map(function (f) {
          var obs = ee.Number(f.get('obs'));
          return ee.Feature(null, {
            unit: 'basin', hybas_id: f.get('HYBAS_ID'),
            mean_elev_m: f.get('mean_elev_m'), elev_p90: f.get('elev_p90'),
            Y: Y, window: img.get('w'), doy: img.get('doy'), wdays: img.get('wdays'),
            obs_km2: obs.divide(1e6),
            s40_km2: ee.Number(f.get('s40')).divide(1e6),
            s10_km2: ee.Number(f.get('s10')).divide(1e6),
            snow_frac40: ee.Algorithms.If(obs.gt(0), ee.Number(f.get('s40')).divide(obs), null),
            snow_frac10: ee.Algorithms.If(obs.gt(0), ee.Number(f.get('s10')).divide(obs), null)
          });
        });
      out = out.merge(perB);
    }

    if (EXPORT_BAND_SERIES) {
      var stacked = a.addBands(bandIdx);
      var groups = ee.List(ee.Dictionary(stacked.reduceRegion({
        reducer: ee.Reducer.sum().repeat(3).group({groupField: 3, groupName: 'eb'}),
        geometry: iran, scale: SCALE, maxPixels: 1e12, tileScale: TILE_SCALE
      })).get('groups'));
      var perBand = ee.FeatureCollection(groups.map(function (gg) {
        var d = ee.Dictionary(gg);
        var s = ee.List(d.get('sum'));
        var obs = ee.Number(s.get(0));
        return ee.Feature(null, {
          unit: 'band', eb: d.get('eb'),
          Y: Y, window: img.get('w'), doy: img.get('doy'), wdays: img.get('wdays'),
          obs_km2: obs.divide(1e6),
          s40_km2: ee.Number(s.get(1)).divide(1e6),
          s10_km2: ee.Number(s.get(2)).divide(1e6),
          snow_frac40: ee.Algorithms.If(obs.gt(0), ee.Number(s.get(1)).divide(obs), null),
          snow_frac10: ee.Algorithms.If(obs.gt(0), ee.Number(s.get(2)).divide(obs), null)
        });
      }));
      out = out.merge(perBand);
    }
    return out;
  })).flatten();
}

function scdlscFeatures(Y) {
  var col = ee.ImageCollection(snowYearWindows(Y));
  var Ndays = col.aggregate_sum('wdays');

  function weightedDays(band) {
    return col.map(function (im) {
      return ee.Image(im).select(band).multiply(ee.Number(ee.Image(im).get('wdays')));
    }).sum();
  }
  var observedDays = weightedDays('obs');
  var snowDays40   = weightedDays('s40');
  var snowDays10   = weightedDays('s10');
  var obsOK        = observedDays.gt(0);

  var scd40 = snowDays40.divide(observedDays).multiply(Ndays).updateMask(obsOK).rename('scd40');
  var scd10 = snowDays10.divide(observedDays).multiply(Ndays).updateMask(obsOK).rename('scd10');

  var lsc40 = col.map(function (im) {
      return ee.Image(im).select('s40').multiply(ee.Number(ee.Image(im).get('doy'))).toFloat();
    }).max().updateMask(snowDays40.gt(0)).rename('lsc40');
  var lsc10 = col.map(function (im) {
      return ee.Image(im).select('s10').multiply(ee.Number(ee.Image(im).get('doy'))).toFloat();
    }).max().updateMask(snowDays10.gt(0)).rename('lsc10');

  var obsFrac = observedDays.divide(Ndays).rename('obs_frac');
  var metrics = ee.Image.cat([scd40, scd10, lsc40, lsc10, obsFrac]);

  var out = ee.FeatureCollection([]);

  if (EXPORT_BASIN_SCDLSC) {
    var perB = metrics.reduceRegions({collection: basins, reducer: ee.Reducer.mean(),
                                      scale: SCALE, tileScale: TILE_SCALE})
      .map(function (f) {
        return ee.Feature(null, {
          unit: 'basin', hybas_id: f.get('HYBAS_ID'),
          mean_elev_m: f.get('mean_elev_m'), elev_p90: f.get('elev_p90'), Y: Y,
          scd40: f.get('scd40'), scd10: f.get('scd10'),
          lsc40: f.get('lsc40'), lsc10: f.get('lsc10'), obs_frac: f.get('obs_frac')});
      });
    out = out.merge(perB);
  }

  if (EXPORT_BAND_SCDLSC) {
    var stacked = metrics.addBands(bandIdx);
    var groups = ee.List(ee.Dictionary(stacked.reduceRegion({
      reducer: ee.Reducer.mean().repeat(5).group({groupField: 5, groupName: 'eb'}),
      geometry: iran, scale: SCALE, maxPixels: 1e12, tileScale: TILE_SCALE
    })).get('groups'));
    var perBand = ee.FeatureCollection(groups.map(function (gg) {
      var d = ee.Dictionary(gg);
      var m = ee.List(d.get('mean'));
      return ee.Feature(null, {
        unit: 'band', eb: d.get('eb'), Y: Y,
        scd40: m.get(0), scd10: m.get(1), lsc40: m.get(2), lsc10: m.get(3),
        obs_frac: m.get(4)});
    }));
    out = out.merge(perBand);
  }
  return out;
}

var SERIES_CHUNK = 1;
var SCDLSC_CHUNK = 1;

var SER_SEL = ['unit','hybas_id','eb','mean_elev_m','elev_p90','Y','window','doy','wdays',
               'obs_km2','s40_km2','s10_km2','snow_frac40','snow_frac10'];
var SCD_SEL = ['unit','hybas_id','eb','mean_elev_m','elev_p90','Y',
               'scd40','scd10','lsc40','lsc10','obs_frac'];

var YEARS = [];
for (var Y = Y0; Y <= Y1; Y++) { YEARS.push(Y); }
function chunk(arr, n) { var o = []; for (var i = 0; i < arr.length; i += n) { o.push(arr.slice(i, i + n)); } return o; }

if (EXPORT_BASIN_SERIES || EXPORT_BAND_SERIES) {
  chunk(YEARS, SERIES_CHUNK).forEach(function (cy) {
    var tag = cy[0] + '_' + cy[cy.length - 1];
    var fc = ee.FeatureCollection(cy.map(function (Y) { return seriesFeatures(Y); })).flatten();
    Export.table.toDrive({collection: fc, folder: DRIVE_FOLDER,
      description: 'Iran_snow_series_8day_' + tag,
      fileNamePrefix: 'Iran_snow_series_8day_' + tag, fileFormat: 'CSV', selectors: SER_SEL});
  });
}

if (EXPORT_BASIN_SCDLSC || EXPORT_BAND_SCDLSC) {
  chunk(YEARS, SCDLSC_CHUNK).forEach(function (cy) {
    var tag = cy[0] + '_' + cy[cy.length - 1];
    var fc = ee.FeatureCollection(cy.map(function (Y) { return scdlscFeatures(Y); })).flatten();
    Export.table.toDrive({collection: fc, folder: DRIVE_FOLDER,
      description: 'Iran_snow_SCD_LSC_' + tag,
      fileNamePrefix: 'Iran_snow_SCD_LSC_' + tag, fileFormat: 'CSV', selectors: SCD_SEL});
  });
}

Export.table.toDrive({
  collection: basins.map(function (f) {
    return ee.Feature(null, {hybas_id: f.get('HYBAS_ID'), iran_area_km2: f.get('iran_area_km2'),
      mean_elev_m: f.get('mean_elev_m'), elev_p10: f.get('elev_p10'),
      elev_p50: f.get('elev_p50'), elev_p90: f.get('elev_p90')}); }),
  folder: DRIVE_FOLDER, description: 'Iran_snow_basins_table',
  fileNamePrefix: 'Iran_snow_basins_table', fileFormat: 'CSV',
  selectors: ['hybas_id','iran_area_km2','mean_elev_m','elev_p10','elev_p50','elev_p90']
});

print('Open the Tasks tab and run: series, SCD/LSC, and the basin table.');
