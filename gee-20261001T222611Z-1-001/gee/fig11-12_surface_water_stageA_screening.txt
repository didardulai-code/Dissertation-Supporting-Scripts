var OCCURRENCE_MIN = 25;

var MIN_AREA_KM2      = 5.0;
var MIN_CLEAR_DECADE  = 20;
var MIN_CLEAR_RECENT  = 8;

var FIRST_YEAR = 1991;
var LAST_YEAR  = 2026;

var SCALE        = 30;
var TILE_SCALE   = 4;
var DRIVE_FOLDER = 'iran war project';

var SCENE_CLOUD_MAX = 70;

var gswOcc   = ee.Image('JRC/GSW1_4/GlobalSurfaceWater').select('occurrence');
var gswMonth = ee.ImageCollection('JRC/GSW1_4/MonthlyHistory');

var L5 = ee.ImageCollection('LANDSAT/LT05/C02/T1_L2');
var L7 = ee.ImageCollection('LANDSAT/LE07/C02/T1_L2');
var L8 = ee.ImageCollection('LANDSAT/LC08/C02/T1_L2');
var L9 = ee.ImageCollection('LANDSAT/LC09/C02/T1_L2');

function clearMask(img) {
  var qa = img.select('QA_PIXEL');
  var bad = qa.bitwiseAnd(1 << 1).neq(0)
      .or(qa.bitwiseAnd(1 << 3).neq(0))
      .or(qa.bitwiseAnd(1 << 4).neq(0))
      .or(qa.bitwiseAnd(1 << 5).neq(0));

  var green = img.select(0).mask();
  return bad.not().and(green).rename('clear');
}

function footprint(site) {
  var pt = ee.Geometry.Point([site[4], site[5]]);
  var search = pt.buffer(site[6] * 1000);
  var water = gswOcc.gte(OCCURRENCE_MIN).selfMask().clip(search);

  var areaImg = water.multiply(ee.Image.pixelArea());
  var km2 = ee.Number(areaImg.reduceRegion({
    reducer: ee.Reducer.sum(), geometry: search, scale: SCALE,
    maxPixels: 1e11, tileScale: TILE_SCALE
  }).get('occurrence')).divide(1e6);

  return {geom: search, water: water, km2: km2, pt: pt};
}

function clearCount(coll, geom, y) {
  var start = ee.Date.fromYMD(y, 1, 1);
  var end   = start.advance(1, 'year');
  var c = coll.filterBounds(geom).filterDate(start, end)
              .filter(ee.Filter.lt('CLOUD_COVER', SCENE_CLOUD_MAX));
  var n = c.size();

  var stack = ee.Image(ee.Algorithms.If(
    n.gt(0),
    c.map(clearMask).sum(),
    ee.Image.constant(0).rename('clear')
  ));
  var m = stack.reduceRegion({
    reducer: ee.Reducer.mean(), geometry: geom, scale: SCALE,
    maxPixels: 1e11, tileScale: TILE_SCALE
  }).get('clear');
  return {clear: m, scenes: n};
}

var siteRows = [];
var yearRows = [];

SITES.forEach(function (s) {
  var f = footprint(s);

  var g = f.water.geometry();

  siteRows.push(ee.Feature(null, {
    site: s[0], region: s[1], input: s[2], fn: s[3],
    lon: s[4], lat: s[5], search_km: s[6],
    max_extent_km2: f.km2,
    max_extent_pixels: f.km2.multiply(1e6).divide(SCALE * SCALE).round(),
    passes_size: f.km2.gte(MIN_AREA_KM2),
    terminal: s[3] === 'TERMINAL'
  }));

  for (var y = FIRST_YEAR; y <= LAST_YEAR; y++) {
    var c5 = clearCount(L5, g, y);
    var c7 = clearCount(L7, g, y);
    var c8 = clearCount(L8, g, y);
    var c9 = clearCount(L9, g, y);

    var gm = gswMonth.filterDate(ee.Date.fromYMD(y, 1, 1),
                                 ee.Date.fromYMD(y, 1, 1).advance(1, 'year'))
                     .filterBounds(g).size();

    yearRows.push(ee.Feature(null, {
      site: s[0], region: s[1], year: y,
      clear_L5: c5.clear, clear_L7: c7.clear,
      clear_L8: c8.clear, clear_L9: c9.clear,
      scenes_L5: c5.scenes, scenes_L7: c7.scenes,
      scenes_L8: c8.scenes, scenes_L9: c9.scenes,
      gsw_months: gm
    }));
  }
});
