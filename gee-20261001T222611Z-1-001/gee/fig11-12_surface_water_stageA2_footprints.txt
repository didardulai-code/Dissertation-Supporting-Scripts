var CLUSTER_SCALE = 120;
var MEASURE_SCALE = 30;
var MIN_AREA_KM2  = 5.0;
var DRIVE_FOLDER  = 'iran war project';
var TILE_SCALE    = 4;

var gsw = ee.Image('JRC/GSW1_4/GlobalSurfaceWater');
var occ = gsw.select('occurrence');
var maxExtent = gsw.select('max_extent');

function largestBody(pt, radius_m) {
  var search = pt.buffer(radius_m);

  var vec = maxExtent.selfMask().reduceToVectors({
    geometry: search,
    scale: CLUSTER_SCALE,
    geometryType: 'polygon',
    eightConnected: true,
    labelProperty: 'w',
    maxPixels: 1e10,
    bestEffort: false
  });

  var withArea = vec.map(function (f) {
    return f.set('a', f.geometry().area(30));
  });

  var n = withArea.size();
  var best = ee.Feature(ee.Algorithms.If(
    n.gt(0),
    withArea.sort('a', false).first(),
    ee.Feature(search, {a: 0})
  ));
  return {geom: best.geometry(), found: n.gt(0)};
}

function areaWithin(mask, geom) {
  return ee.Number(mask.multiply(ee.Image.pixelArea()).reduceRegion({
    reducer: ee.Reducer.sum(), geometry: geom, scale: MEASURE_SCALE,
    maxPixels: 1e11, tileScale: TILE_SCALE
  }).values().get(0)).divide(1e6);
}

var rows = SITES.map(function (s) {
  var pt = ee.Geometry.Point([s[4], s[5]]);
  var b  = largestBody(pt, s[6] * 1000);
  var g  = b.geom;

  var aMax = areaWithin(maxExtent.gt(0), g);
  var a25  = areaWithin(occ.gte(25), g);
  var a50  = areaWithin(occ.gte(50), g);

  var cen = g.centroid(30);

  var offset_km = pt.distance(cen, 30).divide(1000);

  return ee.Feature(null, {
    site: s[0], region: s[1], input: s[2], fn: s[3],
    lon_nominal: s[4], lat_nominal: s[5], search_km: s[6],
    lon_found: ee.Number(cen.coordinates().get(0)),
    lat_found: ee.Number(cen.coordinates().get(1)),
    offset_km: offset_km,
    area_max_km2: aMax,
    area_occ25_km2: a25,
    area_occ50_km2: a50,

    stability: a50.divide(aMax.max(0.0001)),
    passes_size: aMax.gte(MIN_AREA_KM2),
    terminal: s[3] === 'TERMINAL'
  });
});
