function footprint(s) {
  var isTerm = isTerminal(s);
  var pt = ee.Geometry.Point([s[3], s[4]]);
  var search = pt.buffer(s[6] * 1000);

  var identLayer = isTerm ? maxExt.gt(0) : occ.gte(CORE_OCC);

  var blobs = identLayer.selfMask().reduceToVectors({
    geometry: search, scale: VECT_SCALE, geometryType: 'polygon',
    eightConnected: true, labelProperty: 'w', maxPixels: 1e10
  }).map(function (f) {

    return f.set('d', f.geometry().distance(pt, 30))
            .set('a', f.geometry().area(30).divide(1e6));

  }).filter(ee.Filter.gte('a', MIN_CORE_KM2));

  var found = blobs.size().gt(0);
  var core = ee.Feature(ee.Algorithms.If(found,
                                         blobs.sort('d').first(),
                                         ee.Feature(pt.buffer(500))));

  var growLayer = isTerm ? maxExt.gt(0) : occ.gte(FOOT_OCC);

  var grown = growLayer.selfMask().reduceToVectors({
    geometry: search, scale: VECT_SCALE, geometryType: 'polygon',
    eightConnected: true, labelProperty: 'w', maxPixels: 1e10,
    tileScale: TILE_SCALE
  }).filterBounds(core.geometry());

  var n_parts = grown.size();

  var geom = ee.Geometry(ee.Algorithms.If(n_parts.gt(0),
                                          grown.union(30).geometry(),
                                          core.geometry()));
  return {geom: geom, n_parts: n_parts, core: core.geometry(), found: found};
}

function editAnchor(s) {
  var e = EDITS[s[0]];
  return (e.anchor === null) ? [s[3], s[4]] : e.anchor;

function applyEdit(s, geom) {
  var e = EDITS[s[0]];
  if (e.op === 'keep') { return geom; }
  var a = editAnchor(s);
  var disc = ee.Geometry.Point(a).buffer(e.radius_km * 1000, 50);
  var cut  = geom.intersection(disc, 1);
  var parts = ee.FeatureCollection(cut.geometries().map(function (g) {
      return ee.Feature(ee.Geometry(g));
    }))
    .map(function (f) { return f.set('a', f.geometry().area(1).divide(1e6)); })
    .filter(ee.Filter.gte('a', EDIT_MIN_PART_KM2));

  return ee.Geometry(ee.Algorithms.If(parts.size().gt(0), parts.geometry(), cut));
}

var AREA_CRS = 'EPSG:4326';

function num0(v) {
  return ee.Number(ee.List([v, 0]).reduce(ee.Reducer.firstNonNull()));
}

function areaBands(maskImg, geom) {
  return maskImg.multiply(ee.Image.pixelArea()).reduceRegion({
    reducer: ee.Reducer.sum(), geometry: geom,
    crs: AREA_CRS, scale: SCALE,
    maxPixels: 1e11, tileScale: TILE_SCALE
  });
}

function areaKm2(mask, geom) {
  return num0(areaBands(mask, geom).values().get(0)).divide(1e6);
}

    }
  }

  var SITE_FC = ee.FeatureCollection(SITES.map(function (s, i) {
    return ee.Feature(FP[i], {
      site: s[0],
      waterbody_type: s[5],
      core_found: bool01(FPX[i].found)
    });
  }));

  JRC_BLOCKS.forEach(function (blk) {
    var mc = jrcMon.filterDate(blk[0] + '-01-01', (blk[1] + 1) + '-01-01');

    var fc = mc.map(function (img) {
      var band = img.select('water');
      var d = ee.Date(img.get('system:time_start'));

      var stack = band.eq(2).rename('w')
        .addBands(band.gte(1).rename('v'))
        .addBands(ee.Image.constant(1).rename('t'))
        .multiply(ee.Image.pixelArea());

      return stack.reduceRegions({
        collection: SITE_FC,
        reducer: ee.Reducer.sum(),
        crs: AREA_CRS,
        scale: SCALE,
        tileScale: TILE_SCALE
      }).map(function (f) {
        var okJ = ee.Number(f.get('core_found'));
        var wJ = ee.Number(f.get('w')).divide(1e6);
        var vJ = ee.Number(f.get('v')).divide(1e6);
        var tJ = ee.Number(f.get('t')).divide(1e6);
        return ee.Feature(null, {
          site: f.get('site'),
          waterbody_type: f.get('waterbody_type'),
          source: 'JRC',
          year: d.get('year'),
          month: d.get('month'),
          core_found: okJ,
          water_km2:  gateIf(okJ, wJ),
          valid_km2:  gateIf(okJ, vJ),
          total_km2:  tJ,
          valid_frac: gateIf(okJ, vJ.divide(tJ.max(0.0001))),

          water_frac_valid: gateIf(okJ, fracOrNull(wJ, vJ))
        });
      });
    }).flatten();

    Export.table.toDrive({
      collection: fc,
      description: 'Iran_SW_monthly_JRC_' + blk[0] + '_' + blk[1],
      folder: DRIVE_FOLDER,
      fileNamePrefix: 'Iran_SW_monthly_JRC_' + blk[0] + '_' + blk[1],
      fileFormat: 'CSV',
      selectors: ['site','waterbody_type','source','year','month','core_found',
                  'water_km2','valid_km2','total_km2','valid_frac',
                  'water_frac_valid']
    });
  });
}

function prep(coll, greenB, swirB, sensor) {

  return coll.map(function (img) {
    var qa = img.select('QA_PIXEL');

    var bad = qa.bitwiseAnd(1 << 0).neq(0)
        .or(qa.bitwiseAnd(1 << 1).neq(0))
        .or(qa.bitwiseAnd(1 << 2).neq(0))
        .or(qa.bitwiseAnd(1 << 3).neq(0))
        .or(qa.bitwiseAnd(1 << 4).neq(0))
        .or(qa.bitwiseAnd(1 << 5).neq(0));

    var sat = img.select('QA_RADSAT').neq(0);

    var g = img.select(greenB).multiply(0.0000275).add(-0.2);
    var s = img.select(swirB).multiply(0.0000275).add(-0.2);

    var denom = g.add(s);
    var mndwi = g.subtract(s).divide(denom).rename('mndwi');

    var valid = bad.not()
                   .and(sat.not())
                   .and(g.mask())
                   .and(s.mask())
                   .and(denom.abs().gt(0.001));

    var index = mndwi.updateMask(valid);
    var obs   = mndwi.multiply(0).add(1).rename('obs').updateMask(valid);
    return index.addBands(obs)
                .set('sensor', sensor)
                .copyProperties(img, ['system:time_start']);
  });
}

var LS = prep(L5, 'SR_B2', 'SR_B5', 'L5_TM')
  .merge(prep(L7, 'SR_B2', 'SR_B5', 'L7_ETM'))
  .merge(prep(L8, 'SR_B3', 'SR_B6', 'L8_OLI'))
  .merge(prep(L9, 'SR_B3', 'SR_B6', 'L9_OLI'));

function landsatRows(blk, onlySites, sitePred) {
    var use = onlySites
      ? SITES.filter(function (x) { return onlySites.indexOf(x[0]) >= 0; })
      : SITES;
    if (sitePred) { use = use.filter(sitePred); }

    var wantAnnulus = (blk[1] >= ANNULUS_FROM_YEAR);
    var s0 = ee.Date.fromYMD(blk[0], 1, 1);

    var endY = blk[1] + 1;
    var s1 = (endY > 2026) ? ee.Date(SERIES_END_EXCL)
                           : ee.Date.fromYMD(endY, 1, 1);

    var rows = ee.FeatureCollection(use.map(function (site) {

      var i = SITE_INDEX[site[0]];
      var g = FP[i];
      var ok = FPX[i].found;
      var tot = areaKm2(ee.Image.constant(1), g);
      var totOut = wantAnnulus ? areaKm2(ee.Image.constant(1), FPA[i]) : null;

      var coll = LS.filterDate(s0, s1).filterBounds(FPB[i]);

      return coll.map(function (img) {
        var d = ee.Date(img.get('system:time_start'));

        var waterMask = img.select('mndwi').gt(MNDWI_THRESH).unmask(0, false);
        var wM01      = img.select('mndwi').gt(-0.1).unmask(0, false);
        var wP01      = img.select('mndwi').gt(0.1).unmask(0, false);
        var obsMask   = img.select('obs').unmask(0, false);

        var stack = waterMask.rename('w0')
          .addBands(wM01.rename('wm'))
          .addBands(wP01.rename('wp'))
          .addBands(obsMask.rename('ob'));
        var din  = areaBands(stack, g);
        var water = num0(din.get('w0')).divide(1e6);
        var wm01  = num0(din.get('wm')).divide(1e6);
        var wp01  = num0(din.get('wp')).divide(1e6);
        var valid = num0(din.get('ob')).divide(1e6);

        var wOut = null, vOut = null, annOk = null, annSeen = null;
        if (wantAnnulus) {
          var dout = areaBands(stack.select(['w0', 'ob']), FPA[i]);
          wOut = num0(dout.get('w0')).divide(1e6);
          vOut = num0(dout.get('ob')).divide(1e6);

          annSeen = vOut.gt(0);
          annOk = vOut.divide(totOut.max(0.0001)).gte(ANNULUS_MIN_COVERAGE);
        }
        return ee.Feature(null, {
          site: site[0], region: site[1], waterbody_type: site[5],
          sensor: img.get('sensor'),
          date: d.format('YYYY-MM-dd'),
          year: d.get('year'), month: d.get('month'),

          core_found:    bool01(ok),
          water_km2:     gateIf(ok, water),
          water_km2_m01: gateIf(ok, wm01),
          water_km2_p01: gateIf(ok, wp01),
          valid_km2:     gateIf(ok, valid),
          total_km2:     tot,
          valid_frac:    gateIf(ok, valid.divide(tot.max(0.0001))),

          water_frac_valid:     gateIf(ok, fracOrNull(water, valid)),
          water_frac_valid_m01: gateIf(ok, fracOrNull(wm01, valid)),
          water_frac_valid_p01: gateIf(ok, fracOrNull(wp01, valid)),

          annulus_computed: wantAnnulus ? 1 : 0,
          annulus_seen: wantAnnulus ? gateIf(ok, bool01(annSeen)) : null,
          annulus_ok: wantAnnulus ? gateIf(ok, bool01(annOk)) : null,
          water_km2_outside: wantAnnulus
            ? gateIf(ok, ee.Algorithms.If(annOk, wOut, null)) : null,
          valid_km2_outside: wantAnnulus ? gateIf(ok, vOut) : null,
          total_km2_outside: wantAnnulus ? totOut : null,
          valid_frac_outside: wantAnnulus
            ? gateIf(ok, vOut.divide(totOut.max(0.0001))) : null
        });
      });
    })).flatten();

  return rows;
}

var INV_TOL = 1.001;

var INV_CHECK_SITES = ['Lake Urmia', 'Latyan', 'Dez'];
function invariantReport(rows, label) {

  var m = rows.filter(ee.Filter.eq('core_found', 1)).map(function (f) {
    var t = ee.Number(f.get('total_km2')).max(0.0001);
    var v = ee.Number(f.get('valid_km2'));
    var w = ee.Number(f.get('water_km2'));
    var vt = v.divide(t);
    var wv = w.divide(v.max(0.0001));
    var vf = ee.Number(f.get('valid_frac'));
    return f.set({
      valid_total_ratio: vt,
      water_valid_ratio: wv,
      one: 1,
      bad_valid_gt_total: vt.gt(INV_TOL).multiply(1),
      bad_water_gt_valid: wv.gt(INV_TOL).multiply(1),
      bad_frac_range: vf.lt(0).or(vf.gt(INV_TOL)).multiply(1)
    });
  });

  var sums = ee.Dictionary(m.reduceColumns(
    ee.Reducer.sum().repeat(4),
    ['one', 'bad_valid_gt_total', 'bad_water_gt_valid', 'bad_frac_range']
  )).get('sum');
  var sl = ee.List(sums);

  var rng = ee.Dictionary(m.reduceColumns(
    ee.Reducer.minMax().repeat(3),
    ['valid_frac', 'valid_total_ratio', 'water_valid_ratio']
  ));
  var mn = ee.List(rng.get('min'));
  var mx = ee.List(rng.get('max'));

  var summary = {};
  summary.check = label;
  summary.rows_core_found_1 = sl.get(0);
  summary.violation_valid_gt_total = sl.get(1);
  summary.violation_water_gt_valid = sl.get(2);
  summary.violation_frac_range = sl.get(3);
  summary.valid_frac_min = mn.get(0);
  summary.valid_frac_max = mx.get(0);
  summary.worst_valid_total_ratio = mx.get(1);
  summary.worst_water_valid_ratio = mx.get(2);

  print('===== INVARIANT SUMMARY: ' + label + ' (next line) =====');
  print(ee.Feature(null, summary));
  print('===== END INVARIANT SUMMARY =====');

  var perSite = INV_CHECK_SITES.map(function (nm) {
    return m.filter(ee.Filter.eq('site', nm)).limit(3);
  });
  var samples = perSite[0];
  for (var q = 1; q < perSite.length; q++) { samples = samples.merge(perSite[q]); }
  print(label + ': sample rows, three per site');
  print(samples);
}
