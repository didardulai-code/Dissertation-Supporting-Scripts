var iran   = ee.FeatureCollection('FAO/GAUL/2015/level0')
               .filter(ee.Filter.eq('ADM0_NAME','Iran  (Islamic Republic of)'));
var basins = ee.FeatureCollection('WWF/HydroSHEDS/v1/Basins/hybas_5')
               .filterBounds(iran);

var out = basins.map(function (f) {
  return ee.Feature(f.geometry().simplify(500), {
    hybas_id: f.get('HYBAS_ID'),
    up_area:  f.get('UP_AREA'),
    sub_area: f.get('SUB_AREA')
  });
});

print('basins intersecting Iran:', out.size());
Map.addLayer(out, {color: '018571'}, 'basins');
Map.centerObject(iran, 5);

Export.table.toDrive({
  collection:  out,
  description: 'Iran_hybas5_basins',
  folder:      'GEE_exports',
  fileFormat:  'SHP'
});

Export.table.toDrive({
  collection:  iran.map(function (f) { return ee.Feature(f.geometry().simplify(500), {}); }),
  description: 'Iran_outline',
  folder:      'GEE_exports',
  fileFormat:  'SHP'
});
